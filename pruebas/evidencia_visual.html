SELECT p.nombre_comercial, ib.cantidad_almacenada, b.nombre AS bodega, d.numero_dua, d.estado FROM aduana.inventario_bodega ib JOIN aduana.producto p ON ib.id_producto = p.id_producto JOIN aduana.pais pa ON p.id_pais_origen = pa.id_pais JOIN aduana.declaracion_aduanera d ON ib.id_declaracion = d.id_declaracion JOIN aduana.bodega b ON ib.id_bodega = b.id_bodega WHERE pa.codigo_iso_alfa2 = 'CN' AND d.estado = 'Retenida' AND ib.fecha_salida IS NULL;
